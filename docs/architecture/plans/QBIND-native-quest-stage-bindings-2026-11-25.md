The following standard-library script reproduces the census.

```python
from collections import Counter, defaultdict
from pathlib import Path
import json

plan = json.loads(Path("content/quests/missions/completion-binding-plan.json").read_bytes())
stages, reach, pending, singleton = Counter(), Counter(), Counter(), defaultdict(list)
for quest in plan["records"]:
    kinds = [row["event_identity_associations"]["kind"] for row in quest["stages"]]
    missing = {kind for kind, row in zip(kinds, quest["stages"]) if not row["runtime_enabled"]}
    stages.update(kinds)
    reach.update(set(kinds))
    pending.update(missing)
    objectives = missing - {"complete"}
    if len(objectives) == 1:
        singleton[next(iter(objectives))].append(quest["quest"])
    if not objectives:
        singleton["complete"].append(quest["quest"])
assert plan["counts"] == {"quests": 304, "stages": sum(stages.values())}
print(stages, reach, pending, dict(singleton))
```
